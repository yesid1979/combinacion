-- 1. CONSULTA DE VERIFICACIÓN
SELECT 
    id, 
    contrato_id, 
    numero_cuota, 
    periodo_informe, 
    estado_radicacion
FROM informes_supervision
WHERE estado_radicacion IN ('RADICADA', 'BORRADOR')
  AND (
      periodo_informe LIKE '%-07%' 
      OR periodo_informe LIKE '%-08%' 
      OR periodo_informe LIKE '%-09%'
      OR LOWER(periodo_informe) LIKE '%julio%'
      OR LOWER(periodo_informe) LIKE '%agosto%'
      OR LOWER(periodo_informe) LIKE '%septiembre%'
  )
ORDER BY id DESC;
